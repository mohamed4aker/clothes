======================================================
   MAS POS System - تعليمات التثبيت على السيرفر
======================================================

المتطلبات:
- Ubuntu 22.04 أو أحدث
- صلاحيات Root
- 4GB RAM على الأقل
- 20GB مساحة فارغة

----------------------------------------------------
الخطوة 1: تثبيت .NET 8 Runtime
----------------------------------------------------

apt update && apt upgrade -y
wget https://packages.microsoft.com/config/ubuntu/22.04/packages-microsoft-prod.deb
dpkg -i packages-microsoft-prod.deb
apt update
apt install -y aspnetcore-runtime-8.0


----------------------------------------------------
الخطوة 2: تثبيت SQL Server 2022
----------------------------------------------------

wget -qO- https://packages.microsoft.com/keys/microsoft.asc | tee /etc/apt/trusted.gpg.d/microsoft.asc
add-apt-repository "$(wget -qO- https://packages.microsoft.com/config/ubuntu/22.04/mssql-server-2022.list)"
apt update
apt install -y mssql-server
/opt/mssql/bin/mssql-conf setup

اختار: 1 (Express edition - مجاني)
ادخل كلمة سر قوية وحفظها (هنحتاجها)


----------------------------------------------------
الخطوة 3: ارفع ملفات المشروع
----------------------------------------------------

مهم: السيرفر مش محتاج .NET SDK ولا Visual Studio ولا الكود المصدري.
هو محتاج الملفات المتبنية (published) بس.

طريقة 1 - من GitHub (الأسهل، ومش محتاجة تبني حاجة على جهازك):
  افتح تاب Actions في الريبو -> آخر Build ناجح ->
  انزل تحت لقسم Artifacts -> نزّل mas-web-publish
  ده ZIP جواه التطبيق كامل جاهز للتشغيل.

طريقة 2 - من جهازك:
  dotnet publish src/MAS.Web/MAS.Web.csproj -c Release -o publish

بعدها ارفع محتويات المجلد على السيرفر:
scp -r publish/* root@<server-ip>:/var/www/mas/

أو بأي طريقة تانية (FileZilla, WinSCP, إلخ)

للتأكد إن الرفع تمام: لازم تلاقي ملف MAS.Web.dll جوه /var/www/mas/


----------------------------------------------------
الخطوة 4: عدل ملف appsettings.json
----------------------------------------------------

nano /var/www/mas/appsettings.json

غير الـ connection string لـ:

"DefaultConnection": "Server=localhost;Database=MAS_DB;User Id=sa;Password=<SA_PASSWORD>;TrustServerCertificate=True"

استبدل <SA_PASSWORD> بكلمة السر اللي حطيتها في الخطوة 2


----------------------------------------------------
الخطوة 5: عمل Service لتشغيل النظام تلقائياً
----------------------------------------------------

nano /etc/systemd/system/mas.service

الصق:

[Unit]
Description=MAS POS System
After=network.target mssql-server.service

[Service]
WorkingDirectory=/var/www/mas
ExecStart=/usr/bin/dotnet /var/www/mas/MAS.Web.dll
Restart=always
RestartSec=10
User=root
Environment=ASPNETCORE_ENVIRONMENT=Production
Environment=ASPNETCORE_URLS=http://localhost:5002

[Install]
WantedBy=multi-user.target


احفظ بـ Ctrl+X ثم Y ثم Enter

systemctl daemon-reload
systemctl enable mas
systemctl start mas
systemctl status mas


----------------------------------------------------
الخطوة 6: Nginx + SSL
----------------------------------------------------

apt install -y nginx certbot python3-certbot-nginx

### أ) إعداد الـ WebSocket (مهم جداً - من غيره الموقع مش هيشتغل)

النظام Blazor Server، يعني بيشتغل باتصال WebSocket دايم مع السيرفر.
عشان nginx يمرّر الاتصال ده صح، لازم الأول نضيف الـ map ده.

nano /etc/nginx/conf.d/websocket.conf

الصق:

map $http_upgrade $connection_upgrade {
    default upgrade;
    ''      close;
}


### ب) إعداد الموقع

nano /etc/nginx/sites-available/mas

الصق (استبدل yourdomain.com بالدومين الفعلي):

server {
    listen 80;
    server_name yourdomain.com www.yourdomain.com;

    location / {
        proxy_pass http://localhost:5002;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection $connection_upgrade;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;

        # الاتصال بيفضل مفتوح طول ما الصفحة مفتوحة - من غير كده بيقطع كل دقيقة
        proxy_read_timeout 100s;
        proxy_send_timeout 100s;

        client_max_body_size 50M;
    }
}


>>> تنبيه: النسخ القديمة من الملف ده كان مكتوب فيها
>>>   proxy_set_header Connection keep-alive;
>>> وده بيمنع الـ WebSocket من الاشتغال خالص، والنتيجة صفحة بيضا
>>> أو الموقع بيقطع كل شوية. لازم تكون $connection_upgrade زي فوق.


احفظ ثم:

ln -s /etc/nginx/sites-available/mas /etc/nginx/sites-enabled/
nginx -t
systemctl reload nginx

# تثبيت SSL مجاني
certbot --nginx -d yourdomain.com -d www.yourdomain.com

الخطوة 7: Firewall
----------------------------------------------------

ufw allow 22

ufw allow 443
ufw enable


----------------------------------------------------
الخطوة 8: Backup تلقائي يومي
----------------------------------------------------

mkdir -p /var/backups/mas

crontab -e

ضيف السطر:
0 2 * * * /opt/mssql-tools/bin/sqlcmd -S localhost -U sa -P "<SA_PASSWORD>" -Q "BACKUP DATABASE MAS_DB TO DISK='/var/backups/mas/mas_$(date +\%Y\%m\%d).bak'"
